# Plan 06 — Polish

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** The binary is production-deployable: configuration persists to `~/.chargeghost/config.json`, passwords are stored in the OS keychain, shutdown is graceful with no goroutine leaks, a `/health` endpoint exists, and a multi-stage Dockerfile produces a minimal image.

**Architecture:** Config persistence is added to `internal/config/config.go` as `Load(path)` / `Save(path)`. `POST /api/v1/config/save` now writes to disk and applies pending actions (bridge restart / runtime rebuild). Graceful shutdown propagates a `context.WithCancel` through all goroutines; a `sync.WaitGroup` ensures they all exit before the process terminates. The `/health` endpoint bypasses the engine lock. The Dockerfile uses the official Go builder + `gcr.io/distroless/static` as the runtime image.

**Tech Stack:** Go 1.22, `github.com/zalando/go-keyring` (optional, falls back to env var)

import (
    "encoding/json"
    "errors"
    "os"
    "path/filepath"
)

// Load reads the config from path. Returns DefaultConfig() if the file does not exist.
func Load(path string) (*Config, error) {
    data, err := os.ReadFile(path)
    if err != nil {
        if errors.Is(err, os.ErrNotExist) {
            return DefaultConfig(), nil
        }
        return nil, err
    }
    cfg := DefaultConfig()
    if err := json.Unmarshal(data, cfg); err != nil {
        return nil, err
    }
    return cfg, nil
}

// Save writes the config to path, creating parent directories as needed.
// The password field is not included in the JSON — it is stored via keyring.
func (c *Config) Save(path string) error {
    if err := os.MkdirAll(filepath.Dir(path), 0750); err != nil {
        return err
    }
    data, err := json.MarshalIndent(c, "", "  ")
    if err != nil {
        return err
    }
    return os.WriteFile(path, data, 0600)
}

// DefaultConfigPath returns the platform-standard config file path.
func DefaultConfigPath() string {
    home, _ := os.UserHomeDir()
    return filepath.Join(home, ".chargeghost", "config.json")
}
```

import "github.com/zalando/go-keyring"

const keyringService = "chargeghost"

// GetPassword retrieves the OCPP password from:
//  1. OS keyring (keyed by OCPP ID)
//  2. CHARGEGHOST_PASSWORD environment variable (fallback)
//  3. Returns "" if neither is set.
func GetPassword(ocppID string) string {
    if pw, err := keyring.Get(keyringService, ocppID); err == nil {
        return pw
    }
    return os.Getenv("CHARGEGHOST_PASSWORD")
}

// SetPassword stores the OCPP password in the OS keyring.
func SetPassword(ocppID, password string) error {
    return keyring.Set(keyringService, ocppID, password)
}
```

func SaveConfig(cfg *config.Config) http.HandlerFunc {
    return func(w http.ResponseWriter, r *http.Request) {
        path := config.DefaultConfigPath()

## Task 5: Graceful Shutdown with WaitGroup

**Files:**
- Modify: `cmd/chargeghost/main.go`

- [ ] **Step 1: Add WaitGroup tracking to all goroutines**

Replace `cmd/chargeghost/main.go` with the final version:

```go
package main

import (
    "context"
    "errors"
    "log/slog"
    "net/http"
    "os"
    "os/signal"
    "path/filepath"
    "sync"
    "syscall"
    "time"

    "github.com/chargeghost/engine/internal/api"
    "github.com/chargeghost/engine/internal/config"
    engine "github.com/chargeghost/engine/internal/engine"
    "github.com/chargeghost/engine/internal/ocpp"
    "github.com/chargeghost/engine/internal/ocpp/queue"
    rt "github.com/chargeghost/engine/internal/runtime"
    "github.com/chargeghost/engine/internal/timeline"
    ws "github.com/chargeghost/engine/internal/api/ws"
)

func main() {
    slog.SetDefault(slog.New(slog.NewTextHandler(os.Stdout, &slog.HandlerOptions{
        Level: slog.LevelInfo,
    })))

    // Load config from disk (or use defaults if not found).
    cfgPath := config.DefaultConfigPath()
    cfg, err := config.Load(cfgPath)
    if err != nil {
        slog.Error("failed to load config", "path", cfgPath, "err", err)
        os.Exit(1)
    }
    slog.Info("config loaded", "path", cfgPath, "id", cfg.OCPPID)

    // Inject OS keyring password if set.
    if pw := config.GetPassword(cfg.OCPPID); pw != "" {
        cfg.OCPPPassword = &pw
    }

    // Build engine.
    e := engine.NewEngine(cfg.MultiEVSEMode, cfg.EVBatteryCapacity*1000) // kWh → Wh
    for _, cc := range cfg.Connectors {
        e.AddConnector(cc.Voltage, cc.Current, cc.Phase)
    }

    ctx, cancel := context.WithCancel(context.Background())
    var wg sync.WaitGroup

    // Simulation loop.
    wg.Add(1)
    go func() {
        defer wg.Done()
        rt.NewRuntime(e).Run(ctx)
    }()

    // WebSocket hub.
    hub := ws.NewHub()
    wg.Add(1)
    go func() {
        defer wg.Done()
        hub.Run(ctx)
    }()

    // Tick broadcaster.
    wg.Add(1)
    go func() {
        defer wg.Done()
        ws.StartTicker(ctx, hub, e, 1*time.Second)
    }()

    // OCPP dependencies.
    profileManager := ocpp.NewChargingProfileManager()
    configKeys := ocpp.NewConfigKeyManager()
    authCache := ocpp.NewAuthorizationCache()
    localAuthReal := ocpp.NewLocalAuthListManager()
    timelineStore := timeline.NewStore(1000)

    queuePath := filepath.Join(func() string { h, _ := os.UserHomeDir(); return h }(), ".chargeghost", "message_queue.json")
    messageQueue, err := queue.NewQueue(cfg.PersistMessageQueue, queuePath, 3)
    if err != nil {
        slog.Error("failed to create message queue", "err", err)
        os.Exit(1)
    }

    // Wire firmware/diagnostics callbacks.
    var bridge *ocpp.Bridge // declared early so callbacks can reference it

    fwOnStatus := func(status string) {
        hub.BroadcastMessage(ws.Message{Type: "firmware_status_changed", Data: map[string]string{"status": status}})
        if bridge != nil && bridge.IsConnected() {
            bridge.Dispatcher().Enqueue(ocpp.OCPPCommand{
                Description: "FirmwareStatusNotification",
                Execute:     func() error { return bridge.SendFirmwareStatusNotification(status) },
            })
        }
    }
    diagOnStatus := func(status string) {
        hub.BroadcastMessage(ws.Message{Type: "diagnostics_status_changed", Data: map[string]string{"status": status}})
        if bridge != nil && bridge.IsConnected() {
            bridge.Dispatcher().Enqueue(ocpp.OCPPCommand{
                Description: "DiagnosticsStatusNotification",
                Execute:     func() error { return bridge.SendDiagnosticsStatusNotification(status) },
            })
        }
    }

    firmwareManager := ocpp.NewFirmwareManager(fwOnStatus)
    diagnosticsManager := ocpp.NewDiagnosticsManager(diagOnStatus)
    dataTransferReg := ocpp.NewDataTransferRegistry()

    dispatcher := ocpp.NewCommandDispatcher()
    bridge = ocpp.NewBridge(e, hub, cfg, dispatcher, profileManager, configKeys, authCache, localAuthReal, messageQueue, firmwareManager, diagnosticsManager, dataTransferReg)

    // Wire engine → profile manager limit.
    e.GetLimit = func(connectorID, transactionID int) *float64 {
        session := e.GetSession(connectorID)
        c := e.GetConnector(connectorID)
        if c == nil {
            return nil
        }
        var txStart *time.Time
        if session != nil {
            t := session.StartTime
            txStart = &t
        }
        return profileManager.GetCompositeLimit(connectorID, transactionID, time.Now(), c.Voltage, txStart, c.Phase)
    }

    // Wire engine callbacks.
    e.OnConnectorStatusChanged = func(connectorID int, status engine.ConnectorState) {
        hub.BroadcastMessage(ws.Message{Type: "connector_status_changed", Data: map[string]interface{}{"connector_id": connectorID, "status": string(status)}})
        if bridge.IsConnected() {
            dispatcher.Enqueue(ocpp.OCPPCommand{
                Description: "StatusNotification",
                Execute:     func() error { return bridge.SendStatusNotification(connectorID, "NoError", string(status)) },
            })
        }
    }
    e.OnConnectorParamsChanged = func(connectorID int, voltage, current float64, phase int) {
        hub.BroadcastMessage(ws.Message{Type: "connector_params_changed", Data: map[string]interface{}{"connector_id": connectorID, "voltage": voltage, "current": current, "phase": phase}})
    }
    e.OnSessionStarted = func(connectorID int) {
        session := e.GetSession(connectorID)
        if session == nil || !bridge.IsConnected() {
            return
        }
        idTag := ""
        if session.IDTag != nil {
            idTag = *session.IDTag
        }
        meter, _ := e.GetMeterSnapshot(connectorID)
        resID := session.ReservationID
        dispatcher.Enqueue(ocpp.OCPPCommand{
            Description: "StartTransaction",
            Execute: func() error {
                txID, err := bridge.SendStartTransaction(connectorID, idTag, meter, time.Now(), resID)
                if err != nil {
                    return err
                }
                e.SetActiveTransaction(connectorID, txID)
                return nil
            },
        })
    }
    e.OnSessionStopped = func(connectorID int) {
        info := e.GetLastStoppedSession()
        if info == nil || !bridge.IsConnected() {
            return
        }
        snapshot := *info
        dispatcher.Enqueue(ocpp.OCPPCommand{
            Description: "StopTransaction",
            Execute:     func() error { return bridge.SendStopTransaction(snapshot.MeterStop, time.Now(), snapshot.TransactionID, snapshot.Reason, snapshot.MeterHistory) },
        })
    }
    e.OnReservationExpired = func(reservationID, connectorID int) {
        hub.BroadcastMessage(ws.Message{Type: "reservation_changed", Data: map[string]interface{}{"action": "expired", "reservation_id": reservationID, "connector_id": connectorID}})
    }

    // Start OCPP goroutines.
    wg.Add(1)
    go func() {
        defer wg.Done()
        dispatcher.Run(ctx)
    }()
    wg.Add(1)
    go func() {
        defer wg.Done()
        bridge.Start(ctx)
    }()
    wg.Add(1)
    go func() {
        defer wg.Done()
        ocpp.StartMeterValueTicker(ctx, e, bridge, time.Duration(configKeys.GetMeterValueSampleInterval())*time.Second)
    }()

    // Build and start HTTP server.
    app := &api.AppContext{
        Engine:         e,
        Config:         cfg,
        StartTime:      time.Now(),
        Timeline:       timelineStore,
        LocalAuth:      localAuthReal,
        Firmware:       firmwareManager,
        Diagnostics:    diagnosticsManager,
        Hub:            hub,
        ProfileManager: profileManager,
        ConfigKeys:     configKeys,
        OCPP:           bridge,
    }
    router := api.NewRouter(app)
    srv := api.NewServer(":8080", router)

    wg.Add(1)
    go func() {
        defer wg.Done()
        if err := srv.Start(); err != nil && !errors.Is(err, http.ErrServerClosed) {
            slog.Error("HTTP server error", "err", err)
        }
    }()

    slog.Info("ChargeGhost engine started", "addr", ":8080", "id", cfg.OCPPID)

    // Wait for shutdown signal.
    sig := make(chan os.Signal, 1)
    signal.Notify(sig, syscall.SIGINT, syscall.SIGTERM)
    <-sig

    slog.Info("shutdown signal received — stopping")
    cancel()

    // Shutdown HTTP server gracefully.
    shutCtx, shutCancel := context.WithTimeout(context.Background(), 5*time.Second)
    defer shutCancel()
    _ = srv.Shutdown(shutCtx)

    // Wait for all goroutines.
    wg.Wait()
    slog.Info("all goroutines stopped — goodbye")
}
```

# Stage 1: Build
FROM golang:1.22-alpine AS builder

WORKDIR /build
COPY go.mod go.sum ./
RUN go mod download

COPY . .
RUN CGO_ENABLED=0 GOOS=linux go build -ldflags="-s -w" -o chargeghost ./cmd/chargeghost

# Stage 2: Runtime (distroless for minimal attack surface)
FROM gcr.io/distroless/static:nonroot

COPY --from=builder /build/chargeghost /chargeghost

EXPOSE 8080

ENTRYPOINT ["/chargeghost"]
