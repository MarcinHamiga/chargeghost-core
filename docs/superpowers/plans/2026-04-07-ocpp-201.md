# OCPP 2.0.1 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add comprehensive OCPP 2.0.1 support to ChargGhost, switchable via config, keeping the engine protocol-agnostic.

**Architecture:** Extract a shared `OCPPBridge` interface, move existing 1.6J code into `internal/ocpp/v16/`, build new 2.0.1 adapter in `internal/ocpp/v201/`. Factory in `internal/ocpp/` selects implementation based on `config.OCPPVersion`. Engine unchanged.

**Tech Stack:** Go 1.26+, `lorenzodonini/ocpp-go v0.19.0` (OCPP 2.0.1 types in `ocpp2.0.1/*` packages), `go-chi/chi/v5`, `gorilla/websocket`, `stretchr/testify`

**Spec:** `docs/superpowers/specs/2026-04-07-ocpp-201-design.md`

```go
import (
	"context"

	"github.com/chargeghost/engine/internal/config"
	engine "github.com/chargeghost/engine/internal/engine"
	"github.com/chargeghost/engine/internal/ocpp/queue"

// OCPPBridge is the version-agnostic interface for OCPP communication.
// Implemented by v16.Bridge16 and v201.Bridge201.
type OCPPBridge interface {
	Start(ctx context.Context) error
	Stop()
	IsConnected() bool
	GetHeartbeatInterval() int
	Dispatcher() *CommandDispatcher

	// Outbound messages
	SendBootNotification() error
	SendHeartbeat() error
	SendStatusNotification(connectorID int, errorCode, status string) error
	SendMeterValues(connectorID int, value float64, transactionID int, context string) error
	SendAuthorize(idTag string) error

	// Transaction lifecycle.
	// Returns transaction ID: server-assigned int for 1.6, synthetic int for 2.0.1.
	SendTransactionStart(connectorID int, idTag string, meterStart float64, timestamp time.Time, reservationID *int) (int, error)
	SendTransactionStop(meterStop float64, timestamp time.Time, transactionID int, reason string, meterHistory []engine.MeterRecord) error

	// Firmware/Diagnostics
	SendFirmwareStatusNotification(status string) error
	SendDiagnosticsStatusNotification(status string) error

	// Data transfer
	SendDataTransfer(vendorID, messageID, data string) (string, string, error)
}

// NewBridge creates a version-specific OCPPBridge based on the OCPP version string.
func NewBridge(version string, e *engine.Engine, hub *wsapi.Hub, cfg *config.Config, dispatcher *CommandDispatcher, opts ...interface{}) (OCPPBridge, error) {
	switch version {
	case "1.6":
		// Will be wired in Task 2
		return nil, fmt.Errorf("v16 bridge not yet wired")
	case "2.0.1":
		// Will be wired in Phase 2
		return nil, fmt.Errorf("v201 bridge not yet implemented")
	default:
		return nil, fmt.Errorf("unsupported OCPP version: %s", version)
	}
}
```

```go
package v201

import (
	"context"
	"fmt"
	"log/slog"
	"sync"
	"sync/atomic"
	"time"

	ocpp201 "github.com/lorenzodonini/ocpp-go/ocpp2.0.1"
	"github.com/lorenzodonini/ocpp-go/ocpp2.0.1/availability"
	"github.com/lorenzodonini/ocpp-go/ocpp2.0.1/provisioning"
	"github.com/lorenzodonini/ocpp-go/ocpp2.0.1/types"
	"github.com/lorenzodonini/ocpp-go/ws"

	wsapi "github.com/chargeghost/engine/internal/api/ws"
	"github.com/chargeghost/engine/internal/config"
	engine "github.com/chargeghost/engine/internal/engine"
	ocpp "github.com/chargeghost/engine/internal/ocpp"
	"github.com/chargeghost/engine/internal/ocpp/queue"
)

// Bridge201 implements ocpp.OCPPBridge for OCPP 2.0.1.
type Bridge201 struct {
	cs           ocpp201.ChargingStation
	wsClient     *ws.Client
	dispatcher   *ocpp.CommandDispatcher
	engine       *engine.Engine
	hub          *wsapi.Hub
	cfg          *config.Config
	authCache    *ocpp.AuthorizationCache
	localAuth    ocpp.LocalAuthManager
	queue        queue.MessageQueue
	fwManager    ocpp.FirmwareManager
	diagManager  ocpp.DiagnosticsManager
	dataTransfer *ocpp.DataTransferRegistry
	connected    atomic.Bool
	heartbeatInt int

	// Transaction event builders, keyed by EVSE ID
	mu         sync.Mutex
	txBuilders map[int]*TransactionEventBuilder
	nextTxInt  int // synthetic int transaction ID for engine compatibility
}

func NewBridge(e *engine.Engine, hub *wsapi.Hub, cfg *config.Config, dispatcher *ocpp.CommandDispatcher) *Bridge201 {
	b := &Bridge201{
		engine:       e,
		hub:          hub,
		cfg:          cfg,
		dispatcher:   dispatcher,
		heartbeatInt: 300,
		txBuilders:   make(map[int]*TransactionEventBuilder),
	}
	return b
}

// SetManagers sets optional shared managers. Called after construction when
// managers need back-references to the bridge.
func (b *Bridge201) SetManagers(authCache *ocpp.AuthorizationCache, la ocpp.LocalAuthManager, q queue.MessageQueue, fw ocpp.FirmwareManager, diag ocpp.DiagnosticsManager, dt *ocpp.DataTransferRegistry) {
	b.authCache = authCache
	b.localAuth = la
	b.queue = q
	b.fwManager = fw
	b.diagManager = diag
	b.dataTransfer = dt
}

func (b *Bridge201) Start(ctx context.Context) error {
	wsClient := ws.NewClient()
	wsClient.SetDisconnectedHandler(func(err error) {
		slog.Warn("OCPP 2.0.1 disconnected", "error", err)
		b.connected.Store(false)
		if b.hub != nil {
			b.hub.BroadcastMessage(wsapi.Message{
				Type: "connection_state_changed",
				Data: map[string]bool{"connected": false},
			})
		}
	})
	wsClient.SetReconnectedHandler(func() {
		slog.Info("OCPP 2.0.1 reconnected")
		b.connected.Store(true)
		if b.hub != nil {
			b.hub.BroadcastMessage(wsapi.Message{
				Type: "connection_state_changed",
				Data: map[string]bool{"connected": true},
			})
		}
		if b.queue != nil {
			go b.drainQueue()
		}
		b.dispatcher.Enqueue(ocpp.OCPPCommand{
			Description: "BootNotification",
			Execute:     b.SendBootNotification,
		})
	})

	b.wsClient = wsClient
	b.cs = ocpp201.NewChargingStation(b.cfg.OCPPID, nil, wsClient)

	// Register handlers
	b.cs.SetProvisioningHandler(b)
	b.cs.SetAvailabilityHandler(b)
	// More handlers registered in later tasks

	go b.heartbeatLoop(ctx)

	url := b.cfg.ConnectionURL
	slog.Info("connecting to CSMS (OCPP 2.0.1)", "url", url)
	return b.cs.Start(url)
}

func (b *Bridge201) Stop() {
	b.cs.Stop()
}

func (b *Bridge201) IsConnected() bool {
	return b.connected.Load()
}

func (b *Bridge201) GetHeartbeatInterval() int {
	return b.heartbeatInt
}

func (b *Bridge201) Dispatcher() *ocpp.CommandDispatcher {
	return b.dispatcher
}

func (b *Bridge201) heartbeatLoop(ctx context.Context) {
	for {
		select {
		case <-ctx.Done():
			return
		case <-time.After(time.Duration(b.heartbeatInt) * time.Second):
			if b.IsConnected() {
				b.dispatcher.Enqueue(ocpp.OCPPCommand{
					Description: "Heartbeat",
					Execute:     b.SendHeartbeat,
				})
			}
		}
	}
}

func (b *Bridge201) drainQueue() {
	for b.queue != nil {
		msg, ok := b.queue.Peek()
		if !ok {
			return
		}
		// Re-send based on message type
		slog.Info("draining queued message", "type", msg.Type, "id", msg.ID)
		b.queue.Dequeue(msg.ID)
	}
}
```

```go
func (b *Bridge201) OnRequestStopTransaction(request *remotecontrol.RequestStopTransactionRequest) (*remotecontrol.RequestStopTransactionResponse, error) {
	slog.Info("RequestStopTransaction received", "txId", request.TransactionID)

	// Find EVSE by transaction ID string
	b.mu.Lock()
	var evseID int
	found := false
	for eid, builder := range b.txBuilders {
		if builder.TransactionID() == request.TransactionID {
			evseID = eid
			found = true
			break
		}
	}
	b.mu.Unlock()

	if !found {
		return remotecontrol.NewRequestStopTransactionResponse(remotecontrol.RequestStartStopStatusRejected), nil
	}

	err := b.engine.StopSession(evseID, "Remote")
	if err != nil {
		return remotecontrol.NewRequestStopTransactionResponse(remotecontrol.RequestStartStopStatusRejected), nil
	}

	return remotecontrol.NewRequestStopTransactionResponse(remotecontrol.RequestStartStopStatusAccepted), nil
}

func (b *Bridge201) OnTriggerMessage(request *remotecontrol.TriggerMessageRequest) (*remotecontrol.TriggerMessageResponse, error) {
	slog.Info("TriggerMessage received", "requestedMessage", request.RequestedMessage)

	switch request.RequestedMessage {
	case remotecontrol.MessageTriggerBootNotification:
		b.dispatcher.Enqueue(ocpp.OCPPCommand{Description: "BootNotification (triggered)", Execute: b.SendBootNotification})
	case remotecontrol.MessageTriggerHeartbeat:
		b.dispatcher.Enqueue(ocpp.OCPPCommand{Description: "Heartbeat (triggered)", Execute: b.SendHeartbeat})
	case remotecontrol.MessageTriggerStatusNotification:
		evseID := 1
		if request.Evse != nil {
			evseID = request.Evse.ID
		}
		status := b.engine.GetConnectorStatus(evseID)
		b.dispatcher.Enqueue(ocpp.OCPPCommand{
			Description: "StatusNotification (triggered)",
			Execute:     func() error { return b.SendStatusNotification(evseID, "", status) },
		})
	case remotecontrol.MessageTriggerTransactionEvent:
		// Send an Updated event for the active transaction if one exists
		slog.Info("TransactionEvent trigger — sending meter update if active")
	case remotecontrol.MessageTriggerFirmwareStatusNotification:

				Description: "FirmwareStatusNotification (triggered)",
				Execute:     func() error { return b.SendFirmwareStatusNotification(status) },
```

```go
func (b *Bridge201) drainQueue() {
	for b.queue != nil && b.IsConnected() {
		msg, ok := b.queue.Peek()
		if !ok {
			return
		}

		var sendErr error
		switch msg.Type {
		case "TransactionEvent":
			if req, ok := msg.Payload.(*transactions.TransactionEventRequest); ok {
				cb := func(resp *transactions.TransactionEventResponse, err error) {
					if err != nil {
						slog.Error("queued TransactionEvent failed", "error", err)
					}
				}
				sendErr = b.cs.SendRequestAsync(req, cb)
			}
		default:
			slog.Warn("unknown queued message type", "type", msg.Type)
		}

		if sendErr != nil {
			slog.Error("failed to drain queued message", "type", msg.Type, "error", sendErr)
			return
		}
		b.queue.Dequeue(msg.ID)
	}
}
```

```go
package v201

import (
	"sync"

	"github.com/lorenzodonini/ocpp-go/ocpp2.0.1/provisioning"
	ocpp201types "github.com/lorenzodonini/ocpp-go/ocpp2.0.1/types"
)

type MutabilityType string

const (
	MutabilityReadOnly  MutabilityType = "ReadOnly"
	MutabilityReadWrite MutabilityType = "ReadWrite"
	MutabilityWriteOnly MutabilityType = "WriteOnly"
)

type componentVariableKey struct {
	Component string
	Instance  string
	Variable  string
}

type variableEntry struct {
	Value      string
	Mutability MutabilityType
}

type GetVariableResultInternal struct {
	Value  string
	Status provisioning.GetVariableStatus
}

// DeviceModel stores OCPP 2.0.1 Component/Variable data.
type DeviceModel struct {
	mu        sync.RWMutex
	variables map[componentVariableKey]variableEntry
	engine    interface{} // EngineView, for dynamic values — wired later
}

func NewDeviceModel(engineView interface{}) *DeviceModel {
	return &DeviceModel{
		variables: make(map[componentVariableKey]variableEntry),
		engine:    engineView,
	}
}

// SetVariable sets a variable internally (used at startup and by engine).
func (dm *DeviceModel) SetVariable(component, instance, variable, value string, mutability MutabilityType) {
	dm.mu.Lock()
	defer dm.mu.Unlock()
	key := componentVariableKey{Component: component, Instance: instance, Variable: variable}
	dm.variables[key] = variableEntry{Value: value, Mutability: mutability}
}

// SetVariableExternal handles CSMS SetVariables requests. Rejects read-only.
func (dm *DeviceModel) SetVariableExternal(component, instance, variable, value string) provisioning.SetVariableStatus {
	dm.mu.Lock()
	defer dm.mu.Unlock()
	key := componentVariableKey{Component: component, Instance: instance, Variable: variable}

	entry, ok := dm.variables[key]
	if !ok {
		return provisioning.SetVariableStatusUnknownComponent
	}
	if entry.Mutability == MutabilityReadOnly {
		return provisioning.SetVariableStatusRejected
	}

	entry.Value = value
	dm.variables[key] = entry
	return provisioning.SetVariableStatusAccepted
}

// GetVariable returns a variable value or an error status.
func (dm *DeviceModel) GetVariable(component, instance, variable string) GetVariableResultInternal {
	dm.mu.RLock()
	defer dm.mu.RUnlock()
	key := componentVariableKey{Component: component, Instance: instance, Variable: variable}

	entry, ok := dm.variables[key]
	if !ok {
		return GetVariableResultInternal{Status: provisioning.GetVariableStatusUnknownComponent}
	}
	return GetVariableResultInternal{Value: entry.Value, Status: provisioning.GetVariableStatusAccepted}
}

// AllVariables returns all stored variables for reporting.
func (dm *DeviceModel) AllVariables() []struct {
	Component  string
	Instance   string
	Variable   string
	Value      string
	Mutability MutabilityType
} {
	dm.mu.RLock()
	defer dm.mu.RUnlock()

	var result []struct {
		Component  string
		Instance   string
		Variable   string
		Value      string
		Mutability MutabilityType
	}
	for key, entry := range dm.variables {
		result = append(result, struct {
			Component  string
			Instance   string
			Variable   string
			Value      string
			Mutability MutabilityType
		}{
			Component:  key.Component,
			Instance:   key.Instance,
			Variable:   key.Variable,
			Value:      entry.Value,
			Mutability: entry.Mutability,
		})
	}
	return result
}

// BuildGetVariablesResponse processes a GetVariablesRequest.
func (dm *DeviceModel) BuildGetVariablesResponse(data []provisioning.GetVariableData) []provisioning.GetVariableResult {
	results := make([]provisioning.GetVariableResult, len(data))
	for i, d := range data {
		r := dm.GetVariable(d.Component.Name, d.Component.Instance, d.Variable.Name)
		results[i] = provisioning.GetVariableResult{
			AttributeStatus: r.Status,
			AttributeValue:  r.Value,
			Component:       d.Component,
			Variable:        d.Variable,
		}
	}
	return results

// BuildSetVariablesResponse processes a SetVariablesRequest.
func (dm *DeviceModel) BuildSetVariablesResponse(data []provisioning.SetVariableData) []provisioning.SetVariableResult {
	results := make([]provisioning.SetVariableResult, len(data))
	for i, d := range data {
		status := dm.SetVariableExternal(d.Component.Name, d.Component.Instance, d.Variable.Name, d.AttributeValue)
		results[i] = provisioning.SetVariableResult{
			AttributeStatus: status,
			Component:       d.Component,
			Variable:        d.Variable,
		}
	}
	return results
}

// BuildNotifyReportData builds ReportData for all variables (for GetBaseReport).
func (dm *DeviceModel) BuildNotifyReportData() []provisioning.ReportData {
	all := dm.AllVariables()
	data := make([]provisioning.ReportData, len(all))
	for i, v := range all {
		mut := provisioning.MutabilityReadOnly
		if v.Mutability == MutabilityReadWrite {
			mut = provisioning.MutabilityReadWrite
		} else if v.Mutability == MutabilityWriteOnly {
			mut = provisioning.MutabilityWriteOnly
		}
		data[i] = provisioning.ReportData{
			Component: ocpp201types.Component{Name: v.Component, Instance: v.Instance},
			Variable:  ocpp201types.Variable{Name: v.Variable},
			VariableAttribute: []provisioning.VariableAttribute{
				{
					Value:      v.Value,
					Mutability: mut,
				},
			},
		}
	}
	return data
}
```

```go
package v201

import (
	"log/slog"
	"sync"

	ocpp201types "github.com/lorenzodonini/ocpp-go/ocpp2.0.1/types"
)

// ChargingProfileManager201 manages OCPP 2.0.1 charging profiles.
// Uses the same composite limit algorithm as v16 but stores profiles
// in the 2.0.1 format (EVSE-level, string transactionId).
type ChargingProfileManager201 struct {
	mu       sync.RWMutex
	profiles map[int]ocpp201types.ChargingProfile // keyed by profile ID
}

func NewChargingProfileManager201() *ChargingProfileManager201 {
	return &ChargingProfileManager201{
		profiles: make(map[int]ocpp201types.ChargingProfile),
	}
}

func (pm *ChargingProfileManager201) SetProfile(profile ocpp201types.ChargingProfile) {
	pm.mu.Lock()
	defer pm.mu.Unlock()
	pm.profiles[profile.ID] = profile
	slog.Info("charging profile set", "id", profile.ID, "purpose", profile.ChargingProfilePurpose, "stackLevel", profile.StackLevel)
}

func (pm *ChargingProfileManager201) ClearProfile(profileID *int, purpose *ocpp201types.ChargingProfilePurposeType, stackLevel *int) int {
	pm.mu.Lock()
	defer pm.mu.Unlock()
	cleared := 0
	for id, p := range pm.profiles {
		if profileID != nil && id != *profileID {
			continue
		}
		if purpose != nil && p.ChargingProfilePurpose != *purpose {
			continue
		}
		if stackLevel != nil && p.StackLevel != *stackLevel {
			continue
		}
		delete(pm.profiles, id)
		cleared++
	}
	return cleared
}

func (pm *ChargingProfileManager201) GetAllProfiles() []ocpp201types.ChargingProfile {
	pm.mu.RLock()
	defer pm.mu.RUnlock()
	result := make([]ocpp201types.ChargingProfile, 0, len(pm.profiles))
	for _, p := range pm.profiles {
		result = append(result, p)
	}
	return result
}
```

```go
package v201

import (
	"fmt"
	"sync"
)

type MonitorType string

const (
	MonitorTypeUpperThreshold MonitorType = "UpperThreshold"
	MonitorTypeLowerThreshold MonitorType = "LowerThreshold"
	MonitorTypeDelta          MonitorType = "Delta"
	MonitorTypePeriodic       MonitorType = "Periodic"
)

type Monitor struct {
	ID           int
	Component    string
	Instance     string
	Variable     string
	Type         MonitorType
	Value        float64
	Severity     int
	LastValue    *float64
}

type MonitoringManager struct {
	mu       sync.RWMutex
	monitors map[int]*Monitor
	nextID   int
	model    *DeviceModel
}

func NewMonitoringManager(model *DeviceModel) *MonitoringManager {
	return &MonitoringManager{
		monitors: make(map[int]*Monitor),
		model:    model,
	}
}

func (mm *MonitoringManager) AddMonitor(component, instance, variable string, monitorType MonitorType, value float64, severity int) (int, error) {
	// Verify variable exists
	result := mm.model.GetVariable(component, instance, variable)
	if result.Status != "Accepted" {
		return 0, fmt.Errorf("unknown variable %s.%s.%s", component, instance, variable)
	}

	mm.mu.Lock()
	defer mm.mu.Unlock()

	mm.nextID++
	m := &Monitor{
		ID:        mm.nextID,
		Component: component,
		Instance:  instance,
		Variable:  variable,
		Type:      monitorType,
		Value:     value,
		Severity:  severity,
	}
	mm.monitors[mm.nextID] = m
	return mm.nextID, nil
}

func (mm *MonitoringManager) ClearMonitor(id int) bool {
	mm.mu.Lock()
	defer mm.mu.Unlock()
	_, ok := mm.monitors[id]
	if ok {
		delete(mm.monitors, id)
	}
	return ok
}

func (mm *MonitoringManager) GetAllMonitors() []*Monitor {
	mm.mu.RLock()
	defer mm.mu.RUnlock()
	result := make([]*Monitor, 0, len(mm.monitors))
	for _, m := range mm.monitors {
		result = append(result, m)
	}
	return result
}
```

```go
package v201

import "sync"

type DisplayMessage struct {
	ID       int
	Priority string
	State    string
	Text     string
	Language string

type DisplayMessageStore struct {
	mu       sync.RWMutex
	messages map[int]DisplayMessage
}

func NewDisplayMessageStore() *DisplayMessageStore {
	return &DisplayMessageStore{
		messages: make(map[int]DisplayMessage),
	}
}

func (s *DisplayMessageStore) Set(msg DisplayMessage) {
	s.mu.Lock()
	defer s.mu.Unlock()
	s.messages[msg.ID] = msg
}

func (s *DisplayMessageStore) Get(id int) (DisplayMessage, bool) {
	s.mu.RLock()
	defer s.mu.RUnlock()
	msg, ok := s.messages[id]
	return msg, ok
}

func (s *DisplayMessageStore) Clear(id int) bool {
	s.mu.Lock()
	defer s.mu.Unlock()
	_, ok := s.messages[id]
	if ok {
		delete(s.messages, id)
	}
	return ok
}

func (s *DisplayMessageStore) GetAll() []DisplayMessage {
	s.mu.RLock()
	defer s.mu.RUnlock()
	result := make([]DisplayMessage, 0, len(s.messages))
	for _, m := range s.messages {
		result = append(result, m)
	}
	return result
}
```

```go
package v201

import "sync"

// CostStore tracks running transaction costs from CSMS CostUpdated messages.
type CostStore struct {
	mu    sync.RWMutex
	costs map[string]float64 // keyed by transaction ID
}

func NewCostStore() *CostStore {
	return &CostStore{costs: make(map[string]float64)}
}

func (cs *CostStore) Update(transactionID string, totalCost float64) {
	cs.mu.Lock()
	defer cs.mu.Unlock()
	cs.costs[transactionID] = totalCost
}

func (cs *CostStore) Get(transactionID string) (float64, bool) {
	cs.mu.RLock()
	defer cs.mu.RUnlock()
	cost, ok := cs.costs[transactionID]
	return cost, ok
}

func (cs *CostStore) Clear(transactionID string) {
	cs.mu.Lock()
	defer cs.mu.Unlock()
	delete(cs.costs, transactionID)
}
```

```go
import (
	"github.com/lorenzodonini/ocpp-go/ocpp2.0.1/localauth"

func (b *Bridge201) OnSendLocalList(request *localauth.SendLocalListRequest) (*localauth.SendLocalListResponse, error) {
	slog.Info("SendLocalList received", "version", request.VersionNumber, "count", len(request.LocalAuthorizationList))
	// Map to shared LocalAuthManager
	if b.localAuth != nil {
		var entries []ocpp.LocalAuthEntry
		for _, item := range request.LocalAuthorizationList {
			entries = append(entries, ocpp.LocalAuthEntry{
				IDTag:  item.IdToken.IdToken,
				Status: string(item.IdTokenInfo.Status),
			})
		}
		updateType := "Full"
		if request.UpdateType == localauth.UpdateTypeDifferential {
			updateType = "Differential"
		}
		err := b.localAuth.UpdateList(request.VersionNumber, entries, updateType)
		if err != nil {
			return localauth.NewSendLocalListResponse(localauth.SendLocalListStatusFailed), nil
		}
	}
	return localauth.NewSendLocalListResponse(localauth.SendLocalListStatusAccepted), nil
}

func (b *Bridge201) OnGetLocalListVersion(request *localauth.GetLocalListVersionRequest) (*localauth.GetLocalListVersionResponse, error) {
	version := 0
	if b.localAuth != nil {
		version = b.localAuth.GetVersion()
	}
	return localauth.NewGetLocalListVersionResponse(version), nil
}
```
